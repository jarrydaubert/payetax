const { chromium, webkit, expect } = require('@playwright/test');
const AxeBuilder = require('@axe-core/playwright').default;
const fs = require('node:fs');
const path = require('node:path');
const output = path.resolve('docs/reports/input-help-2026-10-30/acceptance-375');
fs.mkdirSync(output, { recursive: true });
const observations = [];

async function main() {
  for (const [engine, browserType] of Object.entries({ chromium, webkit })) {
    const browser = await browserType.launch();
    for (const [layout, viewport] of Object.entries({ desktop: { width: 1440, height: 1000 }, mobile: { width: 375, height: 812 } })) {
      const context = await browser.newContext({ viewport, hasTouch: layout === 'mobile', reducedMotion: 'reduce' });
      const page = await context.newPage();
      const consoleMessages = [], pageErrors = [], captures = [], reviewedLabels = [];
      page.on('pageerror', error => pageErrors.push(String(error)));
      page.on('console', message => { if (['error', 'warning'].includes(message.type())) consoleMessages.push(message.text()); });
      await page.addInitScript(() => {
        localStorage.setItem('directorGuideWelcome:dismissed:v1', 'true');
        localStorage.setItem('cookie-consent', JSON.stringify({ necessary: true, analytics: false, version: '1.0' }));
        localStorage.setItem('cookie-consent-timestamp', new Date().toISOString());
      });
      const visible = locator => locator.filter({ visible: true }).first();
      const button = name => visible(page.getByRole('button', { name, exact: true }));
      const screenshot = async (name, target) => {
        if (target) await target.scrollIntoViewIfNeeded();
        const filename = `${engine}-${layout}-${name}.png`;
        await page.screenshot({ path: path.join(output, filename) });
        captures.push(filename);
      };
      const amount = async (locator, value) => { await locator.fill(value); await locator.blur(); };
      await page.goto('http://localhost:3102');
      await page.waitForLoadState('networkidle');
      await expect(page).toHaveTitle(/PayeTax/);
      await page.getByTestId('married-checkbox').check();
      await button('Additional Income Sources').press('Enter');
      await page.getByTestId('add-income-source').press('Enter');
      const payeForm = page.getByRole('heading', { name: 'Enter Income Tax Details', exact: true }).locator('..');
      const fullPaye = `${engine}-${layout}-paye-whole-form.png`;
      await page.evaluate(() => window.scrollTo(0, 0));
      await page.screenshot({ path: path.join(output, fullPaye), fullPage: true, clip: await payeForm.boundingBox() });
      captures.push(fullPaye);
      await expect(payeForm.locator('details')).toHaveCount(0);
      await expect(payeForm.locator('p')).toHaveCount(0);
      reviewedLabels.push(...await payeForm.locator('label').allTextContents());
      const help = button('Help for Pension');
      await help.focus();
      await help.press('Enter');
      await expect(help).toBeFocused();
      await expect(help).toHaveAccessibleDescription(/salary sacrifice/);
      await expect(page.getByRole('region', { name: 'Help for Pension', exact: true })).toBeVisible();
      await screenshot('paye-help');
      const payeAxe = await new AxeBuilder({ page }).include('[data-testid="calculator-section"]').include('[data-radix-popper-content-wrapper]').analyze();
      await help.press('Escape');
      await expect(help).toBeFocused();
      await amount(page.getByTestId('salary-input'), '12x');
      await expect(page.getByTestId('salary-input')).toHaveAttribute('aria-invalid', 'true');
      await expect(page.getByTestId('salary-input')).toHaveAccessibleDescription(/Enter/);
      await screenshot('paye-validation', page.getByTestId('salary-input'));
      await amount(page.getByTestId('salary-input'), '40000');
      await page.getByTestId('tax-code-input').fill('1257L');
      await page.getByTestId('married-checkbox').uncheck();
      await page.getByRole('button', { name: 'Remove income source 1' }).click();
      await page.getByTestId('calculate-button').press('Enter');
      await expect(page.getByTestId('tax-results')).toContainText('32,322.04');
      await amount(page.getByTestId('pension-input'), '5');
      const incomeToggle = page.getByRole('button', { name: 'Additional Income Sources', exact: true });
      if ((await incomeToggle.getAttribute('aria-expanded')) !== 'true') await incomeToggle.press('Enter');
      await page.getByTestId('add-income-source').press('Enter');
      await amount(page.getByTestId('income-source-0-amount'), '100');
      await page.getByTestId('calculate-button').press('Enter');
      await expect(page.getByTestId('pension-method-result-note')).toContainText('take-home may be too high');
      await expect(page.getByTestId('additional-income-result-note')).toBeVisible();
      await screenshot('paye-results-warnings', page.getByTestId('pension-method-result-note'));
      await screenshot('paye-additional-income-warning', page.getByTestId('additional-income-result-note'));
      const resultHelpNames = [];
      const resultButtons = page.getByRole('region', { name: 'Tax calculation summary results', exact: true }).getByRole('button');
      for (const trigger of await resultButtons.all()) {
        const name = await trigger.innerText();
        resultHelpNames.push(name);
        const activate = () => layout === 'mobile' ? trigger.tap() : trigger.click();
        await trigger.scrollIntoViewIfNeeded();
        await activate();
        await expect(trigger).toHaveAttribute('aria-expanded', 'true');
        await expect(trigger).toHaveAccessibleDescription(/\S/);
        await activate();
        await expect(trigger).toHaveAttribute('aria-expanded', 'false');
        await activate();
        await expect(trigger).toHaveAttribute('aria-expanded', 'true');
        await expect(page.getByRole('region', { name: /^Help for / })).toBeVisible();
        if (layout === 'mobile') await page.getByTestId('salary-input').tap();
        else await page.getByTestId('salary-input').click();
        await expect(trigger).toHaveAttribute('aria-expanded', 'false');
        await trigger.focus();
        await trigger.press('Enter');
        await expect(trigger).toBeFocused();
        await expect(trigger).toHaveAttribute('aria-expanded', 'true');
        await trigger.press('Enter');
        await expect(trigger).toHaveAttribute('aria-expanded', 'false');
        await trigger.press('Space');
        await expect(trigger).toHaveAttribute('aria-expanded', 'true');
        await trigger.press('Space');
        await expect(trigger).toHaveAttribute('aria-expanded', 'false');
        await trigger.press('Enter');
        await trigger.press('Escape');
        await expect(trigger).toHaveAttribute('aria-expanded', 'false');
        await expect(trigger).toBeFocused();
      }


      await page.goto('http://localhost:3102/tools/director-guide');
      await expect(page).toHaveTitle(/Director/);
      if (layout === 'mobile') await button('Open calculator inputs').click();
      const directorForm = visible(page.locator('aside').filter({ has: page.getByRole('button', { name: 'Annual', exact: true }) }));
      await expect(directorForm.getByTestId('director-revenue-input')).toBeVisible();
      await screenshot('director-quick-start', directorForm.getByRole('button', { name: 'Annual', exact: true }));
      await amount(directorForm.getByTestId('director-revenue-input'), '100000');
      await amount(directorForm.getByTestId('director-expenses-input'), '20000');
      await directorForm.getByRole('combobox', { name: 'Income Tax Region', exact: true }).click();
      await page.getByRole('option', { name: /England/ }).press('Enter');
      await directorForm.getByRole('combobox', { name: 'Company Year-End', exact: true }).click();
      await page.getByRole('option', { name: 'Other', exact: true }).press('Enter');
      await screenshot('director-company-year-end', directorForm.getByLabel('Custom Year-End Date (MM-DD)', { exact: true }));
      await button('Full Inputs').click();
      await screenshot('director-already-taken', directorForm.getByLabel('YTD Salary', { exact: true }));
      await screenshot('director-situation', directorForm.getByLabel('Other Personal Income', { exact: true }));
      await button('Advanced Options').click();
      await amount(directorForm.getByRole('textbox', { name: 'Employer Pension', exact: true }), '1000');
      await screenshot('director-advanced-pension', directorForm.getByLabel('Employment Allowance', { exact: true }));
      await screenshot('director-advanced-company-car', directorForm.getByLabel('Company Car BIK', { exact: true }));
      await screenshot('director-advanced-salary', directorForm.getByLabel('Minimum Salary', { exact: true }));
      await screenshot('director-compare-setup', directorForm.getByRole('button', { name: 'Apply', exact: true }));
      const annualHelp = await directorForm.locator('button[data-testid^="tooltip-trigger-"]').evaluateAll(elements => elements.map(e => ({ id: e.dataset.testid, name: e.getAttribute('aria-label') })));
      reviewedLabels.push(...await directorForm.locator('label').allTextContents());
      await button('Monthly').click();
      await screenshot('director-monthly-income', directorForm.getByLabel('Monthly Contract Income', { exact: true }));
      await screenshot('director-monthly-cash', directorForm.getByLabel('Minimum Monthly Draw', { exact: true }));
      const monthlyHelp = await directorForm.locator('button[data-testid^="tooltip-trigger-"]').evaluateAll(elements => elements.map(e => ({ id: e.dataset.testid, name: e.getAttribute('aria-label') })));
      reviewedLabels.push(...await directorForm.locator('label').allTextContents());
      await expect(directorForm.locator('[id$="-hint"]')).toHaveCount(0);
      await button('Help for Monthly Business Expenses').focus();
      await button('Help for Monthly Business Expenses').press('Space');
      await expect(button('Help for Monthly Business Expenses')).toBeFocused();
      await expect(page.getByRole('region', { name: 'Help for Monthly Business Expenses', exact: true })).toBeVisible();
      await expect(button('Help for Monthly Business Expenses')).toHaveAccessibleDescription(/monthly business costs/);
      await screenshot('director-help');
      const directorAxe = await new AxeBuilder({ page }).include('aside').include('[data-radix-popper-content-wrapper]').analyze();
      await page.keyboard.press('Escape');
      await expect(button('Help for Monthly Business Expenses')).toBeFocused();
      if (layout === 'mobile') await expect(page.getByRole('dialog', { name: 'Your Numbers', exact: true })).toBeVisible();
      await button('Annual').click();
      await amount(directorForm.getByTestId('director-revenue-input'), '-5');
      await expect(directorForm.getByTestId('director-revenue-input')).toHaveAttribute('aria-invalid', 'true');
      await expect(directorForm.getByTestId('director-revenue-input')).toHaveAccessibleDescription(/Negative/);
      await screenshot('director-validation', directorForm.getByTestId('director-revenue-input'));
      await amount(directorForm.getByTestId('director-revenue-input'), '100000');
      await directorForm.getByTestId('director-includes-vat-checkbox').check();
      await directorForm.getByLabel('Other PAYE employment?', { exact: true }).check();
      if (layout === 'mobile') await button('Close your numbers panel').click();
      await expect(page.getByTestId('director-estimate-notes')).toContainText('Pension tax limits');
      await screenshot('director-results-warnings', page.getByTestId('director-estimate-notes'));
      const simplified = scan => scan.violations.map(v => ({ id: v.id, impact: v.impact, nodes: v.nodes.map(n => n.target) }));
      expect(pageErrors).toEqual([]);
      observations.push({ engine, layout, viewport, reviewedLabels: [...new Set(reviewedLabels)], annualHelp, monthlyHelp, resultHelpNames, directorHelpCount: new Set([...annualHelp, ...monthlyHelp].map(h => h.id)).size, captures, payeAxe: simplified(payeAxe), directorAxe: simplified(directorAxe), pageErrors, consoleMessages });
      fs.writeFileSync(path.join(output, 'whole-form-results.json'), JSON.stringify(observations, null, 2) + '\n');
      console.log(engine, layout, 'whole forms PASS');
      await context.close();
    }
    await browser.close();
  }
}
main().catch(error => { console.error(error.stack); process.exit(1); });
