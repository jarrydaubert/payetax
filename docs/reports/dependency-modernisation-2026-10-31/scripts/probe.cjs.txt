// Drives one server-side capture and one browser page load against a locally built app whose
// NEXT_PUBLIC_SENTRY_DSN points at the local stub. Usage: node probe.cjs <playwrightModuleDir> <baseURL> <outDir>
const fs = require('node:fs');
const path = require('node:path');

const [pwDir, baseURL, outDir] = process.argv.slice(2);
const { chromium } = require(path.join(pwDir, 'playwright'));
fs.mkdirSync(outDir, { recursive: true });

(async () => {
  const res = await fetch(`${baseURL}/api/send-results?email=probe%40example.com&salary=73123`, {
    method: 'POST',
    headers: {
      'content-type': 'application/json',
      origin: 'https://payetax.co.uk',
      referer: 'https://payetax.co.uk/?salary=73123&email=probe%40example.com',
      cookie: 'session=SECRETCOOKIE; cookie-consent=%7B%22analytics%22%3Atrue%7D',
      authorization: 'Bearer SECRETTOKEN',
      'x-forwarded-for': '203.0.113.77',
      'x-real-ip': '203.0.113.77',
      'user-agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) PrivacyProbe/1.0',
    },
    body: JSON.stringify({ email: 'probe@example.com', results: { salary: 73123, netPay: 51234 } }),
  });
  fs.writeFileSync(path.join(outDir, 'api-response.json'), JSON.stringify({ status: res.status, body: await res.text() }));

  const browser = await chromium.launch();
  const context = await browser.newContext({ baseURL });
  await context.addInitScript(() => {
    localStorage.setItem('cookie-consent', JSON.stringify({ analytics: true }));
    localStorage.setItem('cookie-consent-timestamp', new Date().toISOString());
  });
  const page = await context.newPage();
  const tunnel = [];
  page.on('request', (r) => {
    const u = new URL(r.url());
    if (u.pathname.startsWith('/monitoring') || /sentry/i.test(u.hostname)) {
      tunnel.push({ url: r.url(), method: r.method(), body: r.postData() });
    }
  });
  await page.goto('/?salary=73123#tax-calculator');
  await page.waitForLoadState('networkidle');
  await page.getByTestId('salary-input').fill('73123');
  await page.getByTestId('calculate-button').click();
  await page.waitForTimeout(3000);
  await page.goto('/tools/tax-code-decoder');
  await page.waitForLoadState('networkidle');
  await page.waitForTimeout(2000);
  await browser.close();
  fs.writeFileSync(path.join(outDir, 'browser-sentry-requests.json'), JSON.stringify(tunnel, null, 2));
  console.log(`api status ${res.status}; browser sentry requests: ${tunnel.length}`);
})().catch((e) => {
  console.error(e);
  process.exit(1);
});
