// Usage: node capture.cjs <playwrightModuleDir> <baseURL> <outDir>
// Captures full-page screenshots and computed styles for fixed scenarios at 390/768/1280px.
const fs = require('node:fs');
const path = require('node:path');

const [pwDir, baseURL, outDir] = process.argv.slice(2);
const { chromium } = require(path.join(pwDir, 'playwright'));

const WIDTHS = [390, 768, 1280];
const PROPS = [
  'display', 'position', 'top', 'right', 'bottom', 'left', 'z-index', 'box-sizing',
  'width', 'height', 'min-width', 'max-width', 'min-height', 'max-height',
  'margin-top', 'margin-right', 'margin-bottom', 'margin-left',
  'padding-top', 'padding-right', 'padding-bottom', 'padding-left',
  'flex-direction', 'flex-wrap', 'flex-grow', 'flex-shrink', 'flex-basis', 'order',
  'justify-content', 'align-items', 'align-self', 'row-gap', 'column-gap',
  'grid-template-columns', 'grid-template-rows', 'grid-column-start', 'grid-column-end',
  'color', 'background-color', 'background-image', 'opacity', 'visibility',
  'border-top-width', 'border-right-width', 'border-bottom-width', 'border-left-width',
  'border-top-style', 'border-top-color', 'border-bottom-color', 'border-left-color', 'border-right-color',
  'border-top-left-radius', 'border-bottom-right-radius',
  'outline-style', 'outline-width', 'outline-color', 'outline-offset', 'box-shadow',
  'font-family', 'font-size', 'font-weight', 'font-style', 'line-height', 'letter-spacing',
  'text-align', 'text-transform', 'text-decoration-line', 'white-space', 'text-overflow',
  'overflow-x', 'overflow-y', 'transform', 'cursor', 'pointer-events', 'user-select',
  'transition-property', 'transition-duration', 'animation-name', 'font-variant-numeric',
];

async function settle(page) {
  await page.waitForLoadState('networkidle').catch(() => {});
  await page.evaluate(() => document.fonts.ready);
  await page.waitForTimeout(800);
}

async function computed(page) {
  return page.evaluate((props) => {
    const out = {};
    const pathOf = (el) => {
      const parts = [];
      for (let n = el; n && n.nodeType === 1 && n !== document.documentElement; n = n.parentElement) {
        const sib = n.parentElement ? [...n.parentElement.children].filter((c) => c.tagName === n.tagName) : [];
        parts.unshift(`${n.tagName.toLowerCase()}:${sib.indexOf(n)}`);
      }
      return parts.join('>');
    };
    for (const el of document.body.querySelectorAll('*')) {
      if (el.closest('script,style,noscript,template')) continue;
      const cs = getComputedStyle(el);
      const style = {};
      for (const p of props) style[p] = cs.getPropertyValue(p);
      const cls = typeof el.className === 'string' ? el.className : el.getAttribute('class') || '';
      out[pathOf(el)] = { cls: cls.split(/\s+/).filter(Boolean).sort().join(' '), style };
    }
    return out;
  }, PROPS);
}

const scenarios = {
  async calculator(page) {
    await page.goto('/#tax-calculator');
    await settle(page);
  },
  async results(page) {
    await page.goto('/#tax-calculator');
    await settle(page);
    await page.getByTestId('salary-input').fill('35000');
    await page.getByTestId('calculate-button').click();
    await page.getByTestId('results-table').waitFor();
    await settle(page);
  },
  async payslip(page) {
    await page.goto('/#tax-calculator');
    await settle(page);
    await page.getByTestId('salary-input').fill('49131');
    await page.getByTestId('pension-input').fill('4');
    await page.getByTestId('non-taxable-allowances-input').fill('312');
    await page.getByTestId('calculate-button').click();
    await page.getByTestId('results-table').waitFor();
    await settle(page);
  },
  async 'tax-code-decoder'(page) {
    await page.goto('/tools/tax-code-decoder');
    await settle(page);
    await page.getByPlaceholder('e.g., 1257L').fill('S1257L');
    await page.getByRole('button', { name: 'Decode' }).click();
    await settle(page);
  },
  async about(page) {
    await page.goto('/about');
    await settle(page);
  },
  async blog(page) {
    await page.goto('/blog');
    await settle(page);
  },
  async offline(page) {
    await page.goto('/offline');
    await settle(page);
  },
  async 'not-found'(page) {
    await page.goto('/this-page-does-not-exist');
    await settle(page);
  },
  async director(page) {
    await page.goto('/tools/director-guide');
    await settle(page);
  },
};

(async () => {
  fs.mkdirSync(outDir, { recursive: true });
  const browser = await chromium.launch();
  const meta = { browser: browser.version(), baseURL, widths: WIDTHS, scenarios: Object.keys(scenarios) };
  for (const width of WIDTHS) {
    for (const [name, run] of Object.entries(scenarios)) {
      const context = await browser.newContext({
        baseURL,
        viewport: { width, height: 900 },
        deviceScaleFactor: 1,
        reducedMotion: 'reduce',
        locale: 'en-GB',
        timezoneId: 'Europe/London',
      });
      await context.addInitScript(() => {
        if (!localStorage.getItem('cookie-consent')) {
          localStorage.setItem('cookie-consent', JSON.stringify({ analytics: false }));
          localStorage.setItem('cookie-consent-timestamp', new Date().toISOString());
        }
      });
      const page = await context.newPage();
      const errors = [];
      page.on('pageerror', (e) => errors.push(String(e)));
      page.on('console', (m) => m.type() === 'error' && errors.push(m.text()));
      await run(page);
      await page.screenshot({ path: path.join(outDir, `${name}-${width}.png`), fullPage: true });
      fs.writeFileSync(path.join(outDir, `${name}-${width}.styles.json`), JSON.stringify(await computed(page)));
      if (errors.length) fs.writeFileSync(path.join(outDir, `${name}-${width}.errors.txt`), errors.join('\n'));
      await context.close();
    }
  }
  fs.writeFileSync(path.join(outDir, 'meta.json'), JSON.stringify(meta, null, 2));
  await browser.close();
})().catch((e) => {
  console.error(e);
  process.exit(1);
});
