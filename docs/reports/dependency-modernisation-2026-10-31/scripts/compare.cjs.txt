// Usage: node compare.cjs <sharpModuleDir> <dirA> <dirB> [reportFile]
// Compares PNGs pixel-by-pixel and computed-style snapshots element-by-element.
const fs = require('node:fs');
const path = require('node:path');

const [sharpDir, dirA, dirB, reportFile] = process.argv.slice(2);
const sharp = require(path.join(sharpDir, 'sharp'));

async function raw(file) {
  const { data, info } = await sharp(file).ensureAlpha().raw().toBuffer({ resolveWithObject: true });
  return { data, w: info.width, h: info.height };
}

(async () => {
  const lines = [];
  const files = fs.readdirSync(dirA).filter((f) => f.endsWith('.png')).sort();
  for (const f of files) {
    const a = await raw(path.join(dirA, f));
    const b = await raw(path.join(dirB, f));
    let diff = 0;
    if (a.w === b.w && a.h === b.h) {
      for (let i = 0; i < a.data.length; i += 4) {
        if (a.data[i] !== b.data[i] || a.data[i + 1] !== b.data[i + 1] || a.data[i + 2] !== b.data[i + 2]) diff++;
      }
    }
    const size = a.w === b.w && a.h === b.h ? `${a.w}x${a.h}` : `SIZE ${a.w}x${a.h} vs ${b.w}x${b.h}`;
    lines.push(`PNG ${f}: ${size} differingPixels=${a.w === b.w && a.h === b.h ? diff : 'n/a'}`);

    const sa = JSON.parse(fs.readFileSync(path.join(dirA, f.replace('.png', '.styles.json')), 'utf8'));
    const sb = JSON.parse(fs.readFileSync(path.join(dirB, f.replace('.png', '.styles.json')), 'utf8'));
    const keysA = Object.keys(sa);
    const keysB = new Set(Object.keys(sb));
    const onlyA = keysA.filter((k) => !keysB.has(k));
    const onlyB = [...keysB].filter((k) => !(k in sa));
    let clsDiff = 0;
    const styleDiffs = [];
    for (const k of keysA) {
      if (!keysB.has(k)) continue;
      if (sa[k].cls !== sb[k].cls) clsDiff++;
      for (const [p, v] of Object.entries(sa[k].style)) {
        if (sb[k].style[p] !== v) styleDiffs.push(`${k} ${p}: ${v} -> ${sb[k].style[p]}`);
      }
    }
    lines.push(
      `  STYLES elements=${keysA.length}/${keysB.size} onlyA=${onlyA.length} onlyB=${onlyB.length} classSetDiffs=${clsDiff} propDiffs=${styleDiffs.length}`,
    );
    for (const d of styleDiffs.slice(0, 15)) lines.push(`    ${d}`);
    for (const k of onlyA.slice(0, 5)) lines.push(`    onlyA ${k}`);
    for (const k of onlyB.slice(0, 5)) lines.push(`    onlyB ${k}`);
  }
  const text = lines.join('\n');
  if (reportFile) fs.writeFileSync(reportFile, text);
  console.log(text);
})().catch((e) => {
  console.error(e);
  process.exit(1);
});
